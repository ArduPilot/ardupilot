# Call edges that stack_analysis.py treats as possible but that cannot
# happen at runtime, usually because of a runtime check the static
# analysis can't see. Each line is a regular expression for the demangled
# caller and callee, optionally limited to some threads:
#
#   [threads] caller -> callee [via ancestor [through intermediates]] [leaf]
#
# where [threads] is a comma separated list of thread names or categories
# (e.g. UART1 or UART), or [!names] for every thread except those. With
# leaf the callee's own frame is still counted, but not its calls
# With via a rule only applies while a matching ancestor is on the stack.
# With through its condition only follows calls into matching methods;
# leaving that chain restores the ordinary, unsuppressed graph.

# AP_Logger_File::StartNewLogOK() is false in the main thread and
# AP_Logger_Backend::StartNewLogOK() is false outside it, so writes never
# start a new file log. Logs are started from the logger IO thread
^AP_Logger_Backend::WritePrioritisedBlock\( -> ^AP_Logger_File::start_new_log\(

# Storage::_storage_open() returns once storage is open. The first open
# happens when parameters are loaded in the main thread
[!main] ^ChibiOS::Storage::(read_block|write_block|healthy|get_storage_ptr)\( -> ^ChibiOS::Storage::_storage_open\( leaf

# the rcin thread runs the telemetry schedulers of the CRSF, FrSky and
# Spektrum protocol handlers. MSP telemetry is scheduled from the MSP thread
# (AP_MSP::loop -> process_outgoing_data) and for DisplayPort the OSD thread
[rcin] ^AP_RCTelemetry:: -> ^AP_MSP_Telem_Backend::

# ShouldLog() rejects non-main writes until LoggerMessageWriter_DFLogStart
# has emitted every fixed format. Write_MessageChunk writes LOG_MSG_MSG,
# so ensure_format_emitted cannot emit its format in another thread. Keep
# this condition only along the MSG block-writing chain: a dynamic Write()
# or a different packet writer must still be allowed to emit its format.
[!main] ^AP_Logger_Backend::ensure_format_emitted\( -> ^AP_Logger_Backend::Write_Emit_FMT\( via ^AP_Logger_Backend::Write_MessageChunk\( through ^AP_Logger_Backend::(WriteCriticalBlock|WritePrioritisedBlock|ensure_format_emitted)\(

# These workers are entered through the CRSF singleton. It still exists

^AP_CRSF_Telem::get_singleton\( -> ^AP_CRSF_Telem::init\( via ^AP_CRSF_Telem::_(get_telem_data|process_frame)\(
