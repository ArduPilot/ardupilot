# RP2350 Scratch Y registry — SRAM9 (0x20081000, 4 KB), core1 dedicated I-CODE bus.
# Functions here get single-cycle instruction fetch on core1 with zero bus-fabric contention.
# Rule: core1-EXCLUSIVE hot paths only. If core0 also executes a function parked here it
# reaches SRAM9 over the bus fabric and contends with core1's dedicated I-CODE fetch.
# Budget: 3840 B (first 256 B of SRAM9 reserved for c1_vtable, see common_rp2350_smp.ld).
#
# IMPORTANT: functions listed here must NOT also appear in rp2350_ramfunc2_registry.txt.
# The linker claims .text sections first-come-first-served; duplicates would be silently dropped.
#
# Rate controller fast half (core1 only): the rate thread runs rate_controller_run_dt at the
# gyro/rate rate, which calls AC_PID::update_all 3x per tick (roll/pitch/yaw). AC_PID (scalar) is
# used ONLY by the rate controllers - position control uses AC_PID_2D/AC_P_2D/AC_PID_Basic/AC_P_1D,
# so update_all/update_i/get_ff never run on core0.
libraries/AC_AttitudeControl/AC_AttitudeControl_Multi.cpp|AC_AttitudeControl_Multi::rate_controller_run_dt
libraries/AC_PID/AC_PID.cpp|AC_PID::update_all
libraries/AC_PID/AC_PID.cpp|AC_PID::update_i
libraries/AC_PID/AC_PID.cpp|AC_PID::get_ff
#
# Gyro filter fast half (core1 only): apply_gyro_filters runs the harmonic notch and gyro LPF per
# sample, called from AP_InertialSensor_Invensensev3::read_fifo on the SPI0 device-bus thread,
# pinned to core1 (HAL_CORE_SPI0=1). The Vector3 notch apply is reached only through this path.
# Only exercised when INS_HNTCH is configured. HarmonicNotchFilter::update (coeff recalc) stays in
# RAMFUNC2 - it is the slow half and is also reachable from the core0 AP_Vehicle path.
libraries/AP_InertialSensor/AP_InertialSensor_Backend.cpp|AP_InertialSensor_Backend::apply_gyro_filters
libraries/Filter/NotchFilter.h|NotchFilter<Vector3<float>>::apply
libraries/Filter/HarmonicNotchFilter.h|HarmonicNotchFilter<Vector3<float>>::apply
